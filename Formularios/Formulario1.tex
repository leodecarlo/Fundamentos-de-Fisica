\secao{Movimento Balístico}
\begingroup
\footnotesize
\renewcommand{\arraystretch}{0.82}
\[
v_{0x}=v_0\cos\theta,\quad v_{0y}=v_0\sin\theta,\quad g=10\,\tfrac{m}{s^2}
\]

\noindent
\begin{minipage}[t]{0.60\textwidth}
\vspace{0pt}
\textbf{Se $t_0=0$:}\par
\[
\renewcommand{\arraystretch}{1.2}
\begin{array}{@{}ll@{}}
1a)& x(t)=x_0+v_{0x}t=x_0+(v_0\cos\theta)t \\
1b)& y(t)=y_0+v_{0y}t-\tfrac{1}{2}gt^2=y_0+(v_0\sin\theta)t-\tfrac{1}{2}gt^2 \\
2a)& v_x(t)=v_{0x} \\
2b)& v_y(t)=v_{0y}-gt=v_0\sin\theta-gt \\
3)& v_y^2(t)=v_{0y}^2-2g(y(t)-y_0)\quad \leftarrow\ \text{relaciona } v(t)\text{ e }\Delta y \\
4)& t_{\max}=\tfrac{v_{0y}}{g}=\tfrac{v_0\sin\theta}{g}\quad \leftarrow\ \text{instante da altura máx. (relativo ao lançamento)} \\
5)& h_{\max}=\tfrac{v_{0y}^2}{2g}=\tfrac{v_0^2\sin^2\theta}{2g}\quad \leftarrow\ \text{altura máx. (relativa ao lançamento)} \\
6)& R=x(t_f)-x_0=v_{0x}t_f\quad \leftarrow\ \text{alcance em relação ao ponto de lançamento} \\
6')& \text{se } h_f=0:\ R=\tfrac{v_0^2\sin 2\theta}{g}\quad \leftarrow\ \text{lançamento e aterragem à mesma altura} \\
7)& y(x)=y_0+(\tan\theta)(x-x_0)-\tfrac{g(x-x_0)^2}{2(v_0\cos\theta)^2}\quad \leftarrow\ \text{trajetória } y(x) \\
7')& \text{se } (x_0,y_0)=(0,0):\ y(x)=x\tan\theta-\tfrac{gx^2}{2(v_0\cos\theta)^2}
\end{array}
\]
\end{minipage}\hfill
\begin{minipage}[t]{0.38\textwidth}
\vspace{0pt}
\textbf{Se $t_0, x_0, y_0$ são genéricos:}\par
\[
\renewcommand{\arraystretch}{1.3}
\begin{array}{@{}ll@{}}
1a)& x(t)=x_0+v_{0x}(t-t_0) \\
1b)& y(t)=y_0+v_{0y}(t-t_0)-\tfrac{1}{2}g(t-t_0)^2 \\
2a)& v_x(t)=v_{0x} \\
2b)& v_y(t)=v_{0y}-g(t-t_0) \\
3)& v_y^2(t)=v_{0y}^2-2g(y(t)-y_0) \\
4)& t_{\max}=t_0+\tfrac{v_y(t_0)}{g}=t_0+\tfrac{v_0\sin\theta}{g} \\
5)& h_{\max}=\tfrac{v_y^2(t_0)}{2g}=\tfrac{v_0^2\sin^2\theta}{2g} \\
6)& h_f=y(t_f)-y_0,\quad t_f=t_0+\tfrac{v_{0y}+\sqrt{v_{0y}^2-2gh_f}}{g},\quad R=v_{0x}(t_f-t_0) \\
7)& y(x)=y_0+(\tan\theta)(x-x_0)-\tfrac{g(x-x_0)^2}{2(v_0\cos\theta)^2}
\end{array}
\]
\end{minipage}
